; ======================================================================
; DEMO FIXTURE C — "FortiOS style" router (config-system syntax family)
; Unknown to ComplianceForge on purpose — used LIVE in the demo.
; Story: WAN edge router — SSHv2 only but no lockout, no ACL binding,
; cleartext ftp mgmt, debug-level local logging only, telnet admin port open.
; ======================================================================
;
hostname WAN-EDGE-03
;
config system global
    set admin-sport 443
    set admin-telnet-port 23
    set admin-ftp enable
end
;
config system interface
    edit "port1"
        set alias WAN-UPLINK
    next
end
;
config system ssh
    set ssh-version 2
end
;
config system admin
    set admin-password-min-length 4
end
;
config system snmp community
    set community-name "public"
end
;
config log syslogd setting
    set status disable
end
;
config system ntp
    set ntpserver "10.10.1.1"
end
;
config system session-ttl
    set ttl-default 0
end
;
config firewall policy
    set action accept
end
;
; end of config
